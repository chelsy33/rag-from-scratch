## Part 1：Overview（最小 RAG）

```python
from langchain_community.document_loaders import WebBaseLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain_chroma import Chroma
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

# 1) Load
docs = WebBaseLoader("https://lilianweng.github.io/posts/2023-06-23-agent/").load()

# 2) Split
splits = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=100).split_documents(docs)

# 3) Index
vectorstore = Chroma.from_documents(splits, embedding=OpenAIEmbeddings())
retriever = vectorstore.as_retriever(search_kwargs={"k": 4})

# 4) Generate
prompt = ChatPromptTemplate.from_template(
    """只基于上下文回答问题，不知道就说不知道。

上下文:
{context}

问题: {question}
"""
)
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
chain = prompt | llm | StrOutputParser()

question = "What is Task Decomposition?"
context = "\n\n".join([d.page_content for d in retriever.get_relevant_documents(question)])
print(chain.invoke({"context": context, "question": question}))
```

from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

    for d in retriever.get_relevant_documents(q):
        key = d.page_content[:200]
        if key not in seen:
            seen.add(key)
            all_docs.append(d)

print("queries:", queries)
print("unique docs:", len(all_docs))

## Part 6：RAG-Fusion（RRF 融合）

```python
from collections import defaultdict

def reciprocal_rank_fusion(result_lists, k=60):
    scores = defaultdict(float)
    doc_map = {}
    for docs in result_lists:
        for rank, d in enumerate(docs, start=1):
            key = d.page_content[:200]
            scores[key] += 1.0 / (rank + k)
            doc_map[key] = d
    ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
    return [doc_map[key] for key, _ in ranked]

result_lists = [retriever.get_relevant_documents(q) for q in queries]
fused_docs = reciprocal_rank_fusion(result_lists)
print("fused top:", fused_docs[0].page_content[:180])
```

## Part 12：Multi-representation Indexing（多表征索引）

```python
import uuid
from langchain.retrievers.multi_vector import MultiVectorRetriever
from langchain.storage import InMemoryByteStore

# 例：summary 进向量库，原文进 docstore
doc_ids = [str(uuid.uuid4()) for _ in splits]
summary_docs = []
for i, d in enumerate(splits):
    summary_docs.append(type(d)(page_content=d.page_content[:300], metadata={"doc_id": doc_ids[i]}))

summary_store = Chroma.from_documents(summary_docs, OpenAIEmbeddings(), collection_name="multi-repr")
byte_store = InMemoryByteStore()
retriever_mv = MultiVectorRetriever(vectorstore=summary_store, byte_store=byte_store, id_key="doc_id")
retriever_mv.docstore.mset(list(zip(doc_ids, splits)))

print(retriever_mv.get_relevant_documents("agent memory types")[:1])
```
